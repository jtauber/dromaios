# Intel 8255A: mode 0 parallel ports

The 8255A PPI has two eight-bit ports and a third port whose upper and lower
nibbles can have separate directions. The XT uses it for keyboard data,
equipment switches, and board controls. The chip itself does not understand
scan codes or the XT's port addresses. [Intel's datasheet][manual], Figures 6
and 7 and the mode-0 description, defines this profile.

Mode 0 provides latched outputs and unlatched inputs. Modes 1 and 2 require
handshake behavior and are rejected before changing state. A mode-set write
clears all output latches, including when writing the same mode again. The
bit-set/reset form changes one port-C latch bit without changing directions.

## Registers and mode word

Local addresses 0, 1, and 2 select ports A, B, and C; address 3 accepts control
writes. The original 8255A has no control-register readback: that direction
returns `bus-error`, which the board can resolve as an unanswered transfer.

| Mode bit | Meaning |
| --- | --- |
| 7 | 1 selects a mode word; 0 selects bit-set/reset |
| 6:5 | Group A mode; this profile requires 00 |
| 4 | Port A direction: 1 input, 0 output |
| 3 | Upper port C direction |
| 2 | Group B mode; this profile requires 0 |
| 1 | Port B direction |
| 0 | Lower port C direction |

```device
device "i8255a"

state {
  register MODE: 8
  register OUTPUT_A: 8
  register OUTPUT_B: 8
  register OUTPUT_C: 8
  register INPUT_A: 8
  register INPUT_B: 8
  register INPUT_C: 8
}

action clearOutputs "Mode selection resets all output latches" {
  OUTPUT_A <- u8(0)
  OUTPUT_B <- u8(0)
  OUTPUT_C <- u8(0)
}

action resetPorts "Reset selects input on every pin, preserving external levels" {
  MODE <- u8($9B)
  perform clearOutputs()
}

action initialize "Begin reset, with disconnected inputs high" {
  perform resetPorts()
  INPUT_A <- u8($FF)
  INPUT_B <- u8($FF)
  INPUT_C <- u8($FF)
}

source directionA "One bits identify input pins" : 8 {
  mode = register MODE
  return select(bit(mode, 4), u8($FF), u8(0))
}

source directionB "One bits identify input pins" : 8 {
  mode = register MODE
  return select(bit(mode, 1), u8($FF), u8(0))
}

source directionC "Port C has two independently directed nibbles" : 8 {
  mode = register MODE
  return or(select(bit(mode, 3), u8($F0), u8(0)), select(bit(mode, 0), u8($0F), u8(0)))
}

source readPort "Read input pins or output latches according to direction" (input: 8, output: 8, direction: 8): 8 {
  return or(and(input, direction), and(output, xor(direction, u8($FF))))
}

source portA "Port A readback without side effects" : 8 {
  input = register INPUT_A
  output = register OUTPUT_A
  direction = source directionA()
  value = source readPort(input, output, direction)
  return value
}

source portB "Port B readback without side effects" : 8 {
  input = register INPUT_B
  output = register OUTPUT_B
  direction = source directionB()
  value = source readPort(input, output, direction)
  return value
}

source portC "Port C readback without side effects" : 8 {
  input = register INPUT_C
  output = register OUTPUT_C
  direction = source directionC()
  value = source readPort(input, output, direction)
  return value
}

source driveB "Port B outputs, with input drivers released high" : 8 {
  output = register OUTPUT_B
  direction = source directionB()
  return or(output, direction)
}

action setPortInputs "Apply external pin levels together" (a: 8, b: 8, c: 8) {
  INPUT_A <- a
  INPUT_B <- b
  INPUT_C <- c
}

source writeA "Write the port A output latch" (byte: 8): flag {
  OUTPUT_A <- byte
  return 1
}

source writeB "Write the port B output latch" (byte: 8): flag {
  OUTPUT_B <- byte
  return 1
}

source writeC "Write the port C output latch" (byte: 8): flag {
  OUTPUT_C <- byte
  return 1
}

source writeControl "Select mode 0 or set/reset one port C bit" (byte: 8): flag {
  supported: flag = or(not(bit(byte, 7)), zero(and(byte, u8($64))))
  when supported {
    when bit(byte, 7) {
      MODE <- byte
      perform clearOutputs()
    }
    when not(bit(byte, 7)) {
      count = extend(bits(byte, 3, 1), 8)
      mask = iterate(count, u8(1)) {
        return shiftLeft(mask, 0)
      }
      output = register OUTPUT_C
      OUTPUT_C <- or(and(output, xor(mask, u8($FF))), select(bit(byte, 0), mask, u8(0)))
    }
  }
  return supported
}

source validState "Only mode-0 control words are retained" : flag {
  mode = register MODE
  return equal(and(mode, u8($E4)), u8($80))
}

interface I8255a {
  size 4
  initialize initialize
  reset resetPorts
  validate validState
  input setPortInputs
  view portA
  view portB
  view portC
  view driveB
  read 0 portA
  read 1 portB
  read 2 portC
  write 0 writeA
  write 1 writeB
  write 2 writeC
  write 3 writeControl
}
```

Snapshots retain external pin levels and all latches; restoration never resets
the outputs. Disconnected-high inputs and released-high `driveB` are digital
model conventions, not voltage or contention simulation. The board resamples
outputs after **both** data and mode writes and applies its external inputs.
Writing a latch while its pins are inputs does not drive those pins; a later
mode-set write clears that latch rather than exposing the old value.
